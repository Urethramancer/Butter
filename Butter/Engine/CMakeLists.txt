cmake_minimum_required(VERSION 3.20)
project(ButterEngine LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OSX_DEPLOYMENT_TARGET 26.0)
set(OPENSSL_USE_STATIC_LIBS ON)
find_package(LibtorrentRasterbar REQUIRED)
add_library(ButterEngine STATIC ButterEngine.mm)
target_compile_options(ButterEngine PRIVATE -fobjc-arc)
target_link_libraries(ButterEngine PUBLIC LibtorrentRasterbar::torrent-rasterbar "-framework Foundation" "-framework AppKit")
option(BUTTER_ENGINE_TESTS "Build the loopback engine tests" OFF)
if(BUTTER_ENGINE_TESTS)
	add_executable(EngineTests ../../Tests/EngineTests.mm)
	target_include_directories(EngineTests PRIVATE .)
	target_compile_options(EngineTests PRIVATE -fobjc-arc)
	target_link_libraries(EngineTests PRIVATE ButterEngine)
endif()
