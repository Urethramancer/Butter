cmake_minimum_required(VERSION 3.20)
project(ButterEngine LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OSX_DEPLOYMENT_TARGET 26.0)
set(OPENSSL_USE_STATIC_LIBS ON)
find_package(LibtorrentRasterbar REQUIRED)
add_library(ButterEngine STATIC ButterEngine.mm)
target_compile_options(ButterEngine PRIVATE -fobjc-arc)
target_link_libraries(ButterEngine PUBLIC LibtorrentRasterbar::torrent-rasterbar "-framework Foundation" "-framework AppKit")
option(BUTTER_COVERAGE "Instrument only first-party engine code" OFF)
if(BUTTER_COVERAGE)
	target_compile_options(ButterEngine PRIVATE -fprofile-instr-generate -fcoverage-mapping)
	target_link_options(ButterEngine INTERFACE -fprofile-instr-generate)
endif()
set(BUTTER_SANITIZER "" CACHE STRING "First-party sanitizer: address,undefined or thread")
if(BUTTER_SANITIZER)
	if(NOT BUTTER_SANITIZER STREQUAL "address,undefined" AND NOT BUTTER_SANITIZER STREQUAL "thread")
		message(FATAL_ERROR "Unsupported sanitizer")
	endif()
	target_compile_options(ButterEngine PRIVATE -fsanitize=${BUTTER_SANITIZER} -fno-omit-frame-pointer)
	target_link_options(ButterEngine INTERFACE -fsanitize=${BUTTER_SANITIZER})
endif()
option(BUTTER_ENGINE_TESTS "Build the loopback engine tests" OFF)
if(BUTTER_ENGINE_TESTS)
	add_executable(EngineTests ../../Tests/EngineTests.mm)
	target_include_directories(EngineTests PRIVATE .)
	target_compile_options(EngineTests PRIVATE -fobjc-arc)
	target_link_libraries(EngineTests PRIVATE ButterEngine)
	add_executable(InterruptionTests ../../Tests/InterruptionTests.mm)
	target_include_directories(InterruptionTests PRIVATE .)
	target_compile_options(InterruptionTests PRIVATE -fobjc-arc)
	target_link_libraries(InterruptionTests PRIVATE ButterEngine)
	add_executable(EngineDiskFailureTests ../../Tests/EngineDiskFailureTests.mm)
	target_include_directories(EngineDiskFailureTests PRIVATE .)
	target_compile_options(EngineDiskFailureTests PRIVATE -fobjc-arc)
	target_link_libraries(EngineDiskFailureTests PRIVATE ButterEngine)
endif()
